cmake_minimum_required(VERSION 3.20)
project(safe_freeze_candidate_validation LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows executables are required; Linux execution is not native acceptance")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 17)
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
enable_testing()
set(SF_CANDIDATE_ROOT "${CMAKE_SOURCE_DIR}/candidate" CACHE PATH "Adapted upstream tree")
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(disassembler hde64.c)
else()
  set(disassembler hde32.c)
endif()
function(engine target root strict)
  add_library(${target} STATIC "${root}/src/hook.c" "${root}/src/buffer.c"
    "${root}/src/trampoline.c" "${root}/src/hde/${disassembler}")
  target_include_directories(${target} PUBLIC "${root}/include" PRIVATE "${root}/src")
  if(strict)
    target_compile_definitions(${target} PRIVATE MH_SAFE_FREEZE=1)
  endif()
endfunction()
engine(official "${CMAKE_SOURCE_DIR}/inputs/official" FALSE)
engine(candidate_default "${SF_CANDIDATE_ROOT}" FALSE)
engine(candidate_strict "${SF_CANDIDATE_ROOT}" TRUE)
engine(comparison "${CMAKE_SOURCE_DIR}/inputs/comparison" FALSE)
# The comparison fork uses offsetof without including stddef.h itself.
# Supply the standard header through build flags, keeping comparison bytes exact.
if(MSVC)
  target_compile_options(comparison PRIVATE /FIstddef.h)
else()
  target_compile_options(comparison PRIVATE -include stddef.h)
endif()
add_executable(failure_injection "${SF_CANDIDATE_ROOT}/tests/safe_freeze/failure_injection.c")
target_compile_definitions(failure_injection PRIVATE MH_SAFE_FREEZE=1)
target_include_directories(failure_injection PRIVATE "${SF_CANDIDATE_ROOT}/src" "${SF_CANDIDATE_ROOT}/include")
target_link_libraries(failure_injection PRIVATE candidate_strict)
add_test(NAME injected_failure_recovery COMMAND failure_injection)
set_tests_properties(injected_failure_recovery PROPERTIES TIMEOUT 30)
add_executable(real_peer "${SF_CANDIDATE_ROOT}/tests/safe_freeze/real_peer.cpp")
target_link_libraries(real_peer PRIVATE candidate_strict)
add_test(NAME real_suspended_peer COMMAND real_peer)
set_tests_properties(real_suspended_peer PROPERTIES TIMEOUT 45)
add_executable(real_peer_official "${SF_CANDIDATE_ROOT}/tests/safe_freeze/real_peer.cpp")
target_link_libraries(real_peer_official PRIVATE official)
foreach(control IN ITEMS apply remove)
  if(EXISTS "${CMAKE_SOURCE_DIR}/build/test-controls/${control}/src/hook.c")
    add_executable(recovery_control_${control} "${SF_CANDIDATE_ROOT}/tests/safe_freeze/failure_injection.c")
    target_compile_definitions(recovery_control_${control} PRIVATE MH_SAFE_FREEZE=1)
    target_include_directories(recovery_control_${control} PRIVATE
      "${CMAKE_SOURCE_DIR}/build/test-controls/${control}/src"
      "${CMAKE_SOURCE_DIR}/build/test-controls/${control}/include"
      "${SF_CANDIDATE_ROOT}/src")
    target_link_libraries(recovery_control_${control} PRIVATE candidate_strict)
  endif()
endforeach()
if(EXISTS "${CMAKE_SOURCE_DIR}/tests/workload.c")
  foreach(variant IN ITEMS official candidate_default candidate_strict comparison)
    add_executable(workload_${variant} tests/workload.c)
    target_include_directories(workload_${variant} PRIVATE
      "$<IF:$<STREQUAL:${variant},official>,${CMAKE_SOURCE_DIR}/inputs/official/src,$<IF:$<STREQUAL:${variant},comparison>,${CMAKE_SOURCE_DIR}/inputs/comparison/src,${SF_CANDIDATE_ROOT}/src>>")
    target_compile_definitions(workload_${variant} PRIVATE
      SAFE_FREEZE_VARIANT_LABEL="${variant}" SF_CONFIGURATION="$<CONFIG>")
    if(variant STREQUAL candidate_strict)
      target_compile_definitions(workload_${variant} PRIVATE MH_SAFE_FREEZE=1 SF_STRICT=1)
    elseif(variant STREQUAL comparison)
      target_compile_definitions(workload_${variant} PRIVATE SF_COMPARISON=1)
    endif()
    target_link_libraries(workload_${variant} PRIVATE ${variant} psapi advapi32)
  endforeach()
endif()
